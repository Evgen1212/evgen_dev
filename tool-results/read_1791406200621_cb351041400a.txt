   100→   100→function pluralRequests(n: number) {
   101→   101→  const mod10 = n % 10;
   102→   102→  const mod100 = n % 100;
   103→   103→  if (mod10 === 1 && mod100 !== 11) return `${n} заявка`;
   104→   104→  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 12 || mod100 > 14))
   105→   105→    return `${n} заявки`;
   106→   106→  return `${n} заявок`;
   107→   107→}
